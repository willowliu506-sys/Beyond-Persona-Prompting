\newglossaryentry{attention-filter}
{
    name={Attention Filter},
    description={A controlled mechanism that determines which instructional information remains available for subsequent processing}
}

\newglossaryentry{cognitive-process-based-framework}
{
    name={Cognitive-Process-Based Framework},
    description={A factorised learner-simulation framework that represents selected learning differences through explicit information-processing and knowledge-state mechanisms}
}

\newglossaryentry{explicit-long-term-memory}
{
    name={Explicit Long-Term Memory},
    description={An externally inspectable representation of the knowledge encoded by a simulated learner}
}

\newglossaryentry{persona-prompting}
{
    name={Persona Prompting},
    description={The use of natural-language identity or profile instructions to condition the behaviour of a language model}
}

\newglossaryentry{processing-demand}
{
    name={Processing Demand},
    description={A study-specific measure of the processing load associated with an instructional unit}
}

\newglossaryentry{working-memory-capacity}
{
    name={Working-Memory Processing Capacity},
    description={A controlled capacity constraint applied to instructional information before knowledge encoding}
}
